    OUTPUT geometry.hpp
    COMMAND ${CLANG_FORMAT_TOOL}
    COMMAND ${GEN_GEOMETRY_TOOL} geometry.hpp
    DEPENDS
        ${CLANG_FORMAT_TOOL}
        ${GEN_GEOMETRY_TOOL}
    VERBATIM
)

# --- Test Definitions (cc_test) ---

# A macro to simplify test definition as they all follow the same pattern
macro(ADD_CATCH_TEST test_name source_file link_deps)
    add_executable(${test_name} ${source_file})
    target_compile_definitions(${test_name} PRIVATE ${CATCH_DEFINES})
    target_link_libraries(${test_name} PRIVATE ${link_deps} Catch2::Catch2)
    # The Catch2 target auto-registers the test with CTest when the build is configured.
    # If not using the standard Catch2 target, you'd use 'add_test' manually.
endmacro()
